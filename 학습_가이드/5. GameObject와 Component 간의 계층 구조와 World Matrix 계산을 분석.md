  template <typename T>
  std::weak_ptr<T> CGameObject::CreateComponent(
      const std::string& Name,
      const std::string& ParentName = "Root")  // 기본값: "Root"
  {
      std::shared_ptr<CComponent> Component;
      Component.reset(new T);

      Component->SetWorld(mWorld);
      Component->SetOwner(mSelf);
      Component->SetName(Name);
      Component->SetSelf(Component);

      if (!Component->Init())
          return std::weak_ptr<T>();

      if (Component->mComponentType == EComponentType::Scene)
      {
          // ★ 첫 번째 SceneComponent는 자동으로 Root가 됨
          if (mSceneComponentList.empty())
              mRoot = std::dynamic_pointer_cast<CSceneComponent>(Component);

          else
          {
              // ★ ParentName으로 부모 찾기
              if (ParentName == "Root")
              {
                  auto Root = mRoot.lock();
                  if (Root)
                      Root->AddChild(std::dynamic_pointer_cast<CSceneComponent>(Component));
              }
              else
              {
                  // 특정 이름의 부모 컴포넌트 찾기
                  std::shared_ptr<CSceneComponent> Parent;
                  for (size_t i = 0; i < mSceneComponentList.size(); ++i)
                  {
                      if (mSceneComponentList[i]->mName == ParentName)
                      {
                          Parent = mSceneComponentList[i];
                          break;
                      }
                  }

                  if (Parent)
                      Parent->AddChild(...);
                  else
                      Root->AddChild(...);  // 못 찾으면 Root에 붙임
              }
          }

          mSceneComponentList.push_back(...);
      }

      return std::dynamic_pointer_cast<T>(Component);
  }

  4️⃣ 자식 추가 (SceneComponent.cpp:139-152)

  void CSceneComponent::AddChild(std::weak_ptr<CSceneComponent> Child)
  {
      auto _Child = Child.lock();

      if (_Child)
      {
          // 자식의 부모를 자신으로 설정
          _Child->mParent = std::dynamic_pointer_cast<CSceneComponent>(mSelf.lock());

          // 자식 리스트에 추가
          mChildList.push_back(Child);
      }
  }

  5️⃣ 좌표 데이터 구조 (SceneComponent.h:61-85)

  class CSceneComponent
  {
  protected:
      // ★ 상속 플래그
      bool mInheritScale = true;   // 부모 크기 상속 여부
      bool mInheritRot = true;     // 부모 회전 상속 여부

      // ★ Relative 좌표 (부모 기준)
      FVector3 mRelativeScale = FVector3::One;  // (1, 1, 1)
      FVector3 mRelativeRot;                    // (0, 0, 0)
      FVector3 mRelativePos;                    // (0, 0, 0)

      // ★ World 좌표 (월드 기준 절대 좌표)
      FVector3 mWorldScale = FVector3::One;
      FVector3 mWorldRot;
      FVector3 mWorldPos;

      // ★ 변환 행렬
      FMatrix mScaleMatrix;
      FMatrix mRotMatrix;
      FMatrix mTranslateMatrix;
      FMatrix mWorldMatrix;         // 최종 월드 행렬
  };

  좌표 시스템 개념:
  - Relative 좌표: 부모 컴포넌트를 기준으로 한 상대 좌표
  - World 좌표: 월드 원점을 기준으로 한 절대 좌표

  6️⃣ Relative → World 변환 (Scale) (SceneComponent.cpp:331-362)

  void CSceneComponent::InheritScale()
  {
      if (mInheritScale)  // 부모 크기 상속
      {
          auto Parent = mParent.lock();

          if (Parent)
          {
              // World = Relative * Parent's World
              mWorldScale = mRelativeScale * Parent->mWorldScale;
          }
          else  // 부모 없음 (Root)
          {
              mWorldScale = mRelativeScale;
          }
      }
      else  // 상속 안 함
      {
          mWorldScale = mRelativeScale;
      }

      // ★ 자식들에게 재귀적으로 전파
      size_t Size = mChildList.size();
      for (size_t i = 0; i < Size; ++i)
      {
          auto Child = mChildList[i].lock();
          if (Child)
              Child->InheritScale();
      }
  }

  예시:
  부모: WorldScale = (2, 2, 1)
  자식: RelativeScale = (3, 3, 1)
  결과: 자식의 WorldScale = (6, 6, 1)  // 2*3, 2*3, 1*1

  7️⃣ Relative → World 변환 (Rotation) (SceneComponent.cpp:364-415)

  void CSceneComponent::InheritRotation()
  {
      if (mInheritRot)
      {
          auto Parent = mParent.lock();

          if (Parent)
          {
              // World = Relative + Parent's World (각도는 덧셈)
              mWorldRot = mRelativeRot + Parent->mWorldRot;
          }
          else
          {
              mWorldRot = mRelativeRot;
          }
      }
      else
      {
          mWorldRot = mRelativeRot;
      }

      // ★ 회전 행렬로 축(Axis) 계산
      FMatrix RotationMatrix;
      RotationMatrix.Rotation(mWorldRot);

      for (int i = 0; i < EAxis::End; ++i)
      {
          // 기본 축을 회전 행렬로 변환
          mWorldAxis[i] = FVector3::Axis[i].TransformNormal(RotationMatrix);
          mWorldAxis[i].Normalize();
      }

      // ★ 자식들에게 재귀 전파
      for (size_t i = 0; i < Size; ++i)
      {
          auto Child = mChildList[i].lock();
          if (Child)
              Child->InheritRotation();
      }

      // ★ 회전이 바뀌면 위치도 재계산 필요
      for (size_t i = 0; i < Size; ++i)
      {
          auto Child = mChildList[i].lock();
          if (Child)
              Child->InheritPos();
      }
  }

  8️⃣ Relative → World 변환 (Position) (SceneComponent.cpp:417-459)

  void CSceneComponent::InheritPos()
  {
      auto Parent = mParent.lock();

      if (Parent)
      {
          // ★ 부모의 회전과 위치를 담은 변환 행렬 생성
          FMatrix ParentMatrix;

          if (mInheritRot)
          {
              FVector3 ParentRot = Parent->GetWorldRot();
              ParentMatrix.Rotation(ParentRot);
          }

          // 부모의 위치를 행렬의 이동 부분에 설정
          FVector3 ParentPos = Parent->GetWorldPos();
          memcpy(&ParentMatrix._41, &ParentPos, sizeof(FVector3));

          // ★ Relative 좌표를 부모 공간으로 변환
          /*
          예시: 자식이 부모 기준 (3, 0, 0)에 있고
                부모가 월드 (10, 10, 10)에 있을 때

          [3, 0, 0, 1] * [1  0  0  0 ]   = [13, 10, 10, 1]
                         [0  1  0  0 ]
                         [0  0  1  0 ]
                         [10 10 10 1]
          */
          mWorldPos = mRelativePos.TransformCoord(ParentMatrix);
      }
      else  // 부모 없음 (Root)
      {
          mWorldPos = mRelativePos;
      }

      // ★ 자식들에게 재귀 전파
      size_t Size = mChildList.size();
      for (size_t i = 0; i < Size; ++i)
      {
          auto Child = mChildList[i].lock();
          if (Child)
              Child->InheritPos();
      }
  }

  🔟 World Matrix 생성 (SceneComponent.cpp:154-161)

  void CSceneComponent::UpdateTransform()
  {
      // 1. Scale 행렬 생성
      mScaleMatrix.Scaling(mWorldScale);

      // 2. Rotation 행렬 생성
      mRotMatrix.Rotation(mWorldRot);

      // 3. Translation 행렬 생성
      mTranslateMatrix.Translation(mWorldPos);

      // 4. ★ 최종 World Matrix = S * R * T
      mWorldMatrix = mScaleMatrix * mRotMatrix * mTranslateMatrix;
  }

  행렬 곱셈 순서의 의미:
  World = Scale * Rotation * Translation

  1. Scale: 크기 조정
  2. Rotation: 회전
  3. Translation: 이동

  순서가 중요! (S*R*T ≠ T*R*S)

  armComp->SetRelativePos(10, 0, 0);         // Body 기준 (10, 0, 0)

  // 3. 손자 Component 생성 (부모: "Arm")

  [계층 구조]
  ────────────────────────────────────────────────────────────────
  GameObject (player)
    │
    └─ mRoot → Body (CMeshComponent)
                 │  Relative: Pos(100, 100, 0), Scale(1, 1, 1)
                 │  World:    Pos(100, 100, 0), Scale(1, 1, 1)
                 │
                 ├─ Arm (CMeshComponent)
                 │    │  Relative: Pos(10, 0, 0), Scale(1, 1, 1)
                 │    │  World:    Pos(110, 100, 0), Scale(1, 1, 1)
                 │    │
                 │    └─ Hand (CMeshComponent)
                 │         Relative: Pos(5, 0, 0), Scale(0.5, 0.5, 1)
                 │         World:    Pos(115, 100, 0), Scale(0.5, 0.5, 1)
                 │
                 └─ (다른 자식들...)

  CGameObject::PostUpdate(DeltaTime)
    │
    ├─→ Body->PostUpdate(DeltaTime)
    │     └─→ UpdateTransform()

    │           ├─ mRotMatrix.Rotation(mWorldRot)

    │
    ├─→ Arm->PostUpdate(DeltaTime)
    │     └─→ UpdateTransform()

  [좌표 설정 시 전파 프로세스]
  ────────────────────────────────────────────────────────────────

  1. Relative 좌표 설정
     player->SetRelativePos(100, 100, 0);
       └─→ Root->SetRelativePos(100, 100, 0)
             ├─ mRelativePos = (100, 100, 0)
             └─ InheritPos()  ← 자신과 자식들의 World 좌표 갱신


  2. InheritPos() 재귀 호출

     Body (Root):
       ├─ 부모 없음 → mWorldPos = mRelativePos = (100, 100, 0)
       └─ Arm->InheritPos() 호출

     Arm:
       ├─ 부모 있음 (Body)
       │   ├─ ParentMatrix 생성 (Body의 회전 + 위치)
       │   └─ mWorldPos = RelativePos(10,0,0).TransformCoord(ParentMatrix)
       │                = (110, 100, 0)
       └─ Hand->InheritPos() 호출

     Hand:
       ├─ 부모 있음 (Arm)
       │   └─ mWorldPos = RelativePos(5,0,0).TransformCoord(ArmMatrix)
       │                = (115, 100, 0)
       └─ 재귀 종료


  3. Scale 변경 시 전파

     Body->SetRelativeScale(2, 2, 1);
       └─→ InheritScale()
             │
             ├─ Body: mWorldScale = (2, 2, 1)
             │
             ├─→ Arm->InheritScale()
             │     └─ Arm: mWorldScale = (1,1,1) * (2,2,1) = (2, 2, 1)
             │
             └─→ Hand->InheritScale()
                   └─ Hand: mWorldScale = (0.5,0.5,1) * (2,2,1) = (1, 1, 1)


  4. Rotation 변경 시 전파

     Body->SetRelativeRotation(0, 0, 45);  // Z축 45도 회전
       └─→ InheritRotation()
             │
             ├─ Body: mWorldRot = (0, 0, 45)
             │   └─ WorldAxis 계산 (X, Y, Z 축 회전)
             │
             ├─→ Arm->InheritRotation()
             │     ├─ Arm: mWorldRot = (0,0,0) + (0,0,45) = (0, 0, 45)
             │     └─→ Arm->InheritPos()  ← 회전 변경 시 위치도 재계산
             │
             └─→ Hand->InheritRotation()
                   └─→ Hand->InheritPos()

  GameObject의 역할

  // GameObject는 모든 변환을 Root에 위임
  void CGameObject::SetRelativePos(const FVector3& Pos)
  {
      auto Root = mRoot.lock();
      if (Root)
          Root->SetRelativePos(Pos);  // ← 실제 작업은 Root가 함
  }

  GameObject는 단순히 프록시 역할만 합니다!

  좌표 변환 방향

  Relative → World (일반적 케이스):
  SetRelativePos(10, 0, 0);
    └─→ InheritPos() → mWorldPos 계산

  World → Relative (역변환):
  SetWorldPos(100, 100, 0);
    └─→ InheritWorldPos() → mRelativePos 역계산

  재귀적 전파의 효율성

  // 부모 위치 변경 시 모든 자손의 World 좌표 자동 갱신
  Body->SetRelativePos(200, 200, 0);
    └─ InheritPos()
         ├─ Body: World = (200, 200, 0)
         ├─ Arm: World = (210, 200, 0)   ← 자동 갱신
         └─ Hand: World = (215, 200, 0)  ← 자동 갱신

  행렬 변환 순서

  // World Matrix = Scale * Rotation * Translation
  //
  // 정점이 변환되는 순서 (역순):
  // 1. 로컬 좌표 → Translation (이동)
  // 2. → Rotation (회전)
  // 3. → Scale (크기)
  // 4. → 월드 좌표

  // 예시:
  FVector3 localVertex(1, 0, 0);
  FVector3 worldVertex = localVertex.TransformCoord(mWorldMatrix);

  상속 플래그의 활용

  // 부모 회전은 무시하고 위치만 따라가기
