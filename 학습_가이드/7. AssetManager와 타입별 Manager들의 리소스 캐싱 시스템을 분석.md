  bool CTextureManager::LoadTexture(const std::string& Name,
                                     const TCHAR* FileName,
                                     const std::string& PathName)
  {
      // ★ Step 1: 중복 체크 (캐시에 있는지 확인)
      std::weak_ptr<CTexture> Check = FindTexture(Name);

      // ★ 이미 로드된 경우 즉시 반환 (중복 로드 방지!)
      if (!Check.expired())
          return true;

      // ★ Step 2: 새 텍스처 생성
      CTexture* Origin = new CTexture;
      std::shared_ptr<CTexture> Texture;
      Texture.reset(Origin);

      Texture->mName = Name;

      // ★ Step 3: 실제 파일에서 로드
      if (!Texture->LoadTexture(FileName, PathName))
      {
          return false;  // 로드 실패 시 삭제됨 (shared_ptr 자동 해제)
      }

      // ★ Step 4: 캐시에 등록
      mTextureMap.insert(std::make_pair(Name, Texture));

      return true;
  }

  5️⃣ 텍스처 검색 (TextureManager.cpp:139-148)

  std::weak_ptr<CTexture> CTextureManager::FindTexture(const std::string& Name)
  {
      auto iter = mTextureMap.find(Name);

      // ★ 없으면 빈 weak_ptr 반환
      if (iter == mTextureMap.end())
          return std::weak_ptr<CTexture>();

      // ★ 있으면 shared_ptr를 weak_ptr로 변환하여 반환
      return iter->second;
  }

  bool CMeshManager::CreateMesh(const std::string& Name,
                                 void* VertexData, int VertexSize,
                                 int VertexCount, D3D11_USAGE VertexUsage,
                                 D3D11_PRIMITIVE_TOPOLOGY Primitive,

  }

  template <typename T>
  bool CreateShader(const std::string& Name,
                    const std::string& PathName = "Shader")
  {
      // ★ Step 1: 중복 체크
      std::weak_ptr<CShader> Check = FindShader(Name);

      if (!Check.expired())
          return true;

      // ★ Step 2: 타입별 셰이더 생성 (템플릿)
      T* Origin = new T;  // T = CShaderColor2D, CShaderTexture2D 등
      std::shared_ptr<CShader> Shader;
      Shader.reset(Origin);

      // ★ Step 3: HLSL 컴파일
      if (!Shader->Init(PathName))
      {
          return false;
      }

      Shader->SetName(Name);

      // ★ Step 4: 캐시에 등록
      mShaderMap.insert(std::make_pair(Name, Shader));

      return true;
  }

  1️⃣1️⃣ 상수 버퍼 캐싱 (ShaderManager.cpp:55-78)

  bool CShaderManager::CreateCBuffer(const std::string& Name,
                                      int Size, int Register,
                                      int ShaderBuffer)
  {
      // ★ 중복 체크
      std::weak_ptr<CConstantBuffer> Check = FindCBuffer(Name);

      if (!Check.expired())
          return true;

      CConstantBuffer* Origin = new CConstantBuffer;
      std::shared_ptr<CConstantBuffer> CBuffer;
      CBuffer.reset(Origin);

      if (!CBuffer->Init(Size, Register, ShaderBuffer))
          return false;

      CBuffer->SetName(Name);

      // ★ 별도 맵에 캐싱
      mCBufferMap.insert(std::make_pair(Name, CBuffer));

      return true;
  }

  1️⃣2️⃣ 참조 카운트 기반 해제 (TextureManager.cpp:243-255)

  void CTextureManager::ReleaseAsset(const std::string& Name)
  {
      auto iter = mTextureMap.find(Name);

      if (iter != mTextureMap.end())
      {
          // ★ 핵심: use_count() 체크
          // use_count() == 1 → Manager만 소유 (외부 참조 없음)
          if (iter->second.use_count() == 1)
          {
              mTextureMap.erase(iter);  // 캐시에서 제거 → 자동 삭제
          }
          // use_count() > 1 → 외부에서 아직 사용 중 → 유지
      }
  }

  참조 카운트 동작:
  // 초기 상태
  mTextureMap["player"] = shared_ptr<CTexture>(...)  // use_count = 1

  // 외부에서 사용
  std::shared_ptr<CTexture> tex = mTextureMap["player"].lock();  // use_count = 2

  // Release 호출
  ReleaseAsset("player");
  // use_count == 2 → 삭제 안 함 (외부에서 사용 중)

  // 외부 참조 해제
  tex.reset();  // use_count = 1

  // 다시 Release 호출
  ReleaseAsset("player");
  // use_count == 1 → 삭제 진행!

  3. Smart Pointer 기반 메모리 관리

  Manager (소유)
    └─→ shared_ptr<Resource>
           ↑
           │ (weak_ptr)
        사용자 (관찰)

  이점:
  - 자동 메모리 관리
  - 참조 카운팅으로 안전한 해제
  - 댕글링 포인터 방지

  4. unordered_map 선택 이유

  | 자료구조             | 검색     | 삽입     | 삭제     | 메모리 |
  |----------------------|----------|----------|----------|--------|
  | vector               | O(n)     | O(1)     | O(n)     | 작음   |
  | map (RB-Tree)        | O(log n) | O(log n) | O(log n) | 중간   |
  | unordered_map (Hash) | O(1)     | O(1)     | O(1)     | 큼     |

  선택 이유: 리소스 관리는 빠른 검색이 최우선!
