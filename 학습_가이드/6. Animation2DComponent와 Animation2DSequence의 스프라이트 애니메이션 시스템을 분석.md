  7️⃣ Update 함수 (Animation2DSequence.cpp:30-101)

  void CAnimation2DSequence::Update(float DeltaTime)
  {
      if (mAnimation.expired())
          return;

      auto Anim = mAnimation.lock();

      // ★ Step 1: 시간 누적 (재생 속도 적용)
      mTime += DeltaTime * mPlayRate;

      // ★ Step 2: 프레임 전환 판정
      if (mTime >= mFrameTime)
      {
          mTime -= mFrameTime;  // 남은 시간 보존

          // ★ Step 3-A: 역재생
          if (mReverse)
          {
              --mFrame;  // 프레임 감소

              if (mFrame < 0)  // 첫 프레임 이전
              {
                  if (mFinishNotify)
                      mFinishNotify();  // 완료 콜백

                  if (mLoop)
                  {
                      mFrame = Anim->GetFrameCount() - 1;  // 마지막으로

                      // Notify 초기화
                      for (size_t i = 0; i < mNotifyArray.size(); ++i)
                          mNotifyArray[i].Call = false;
                  }
                  else
                      mFrame = 0;  // 첫 프레임 고정
              }
          }

          // ★ Step 3-B: 정방향 재생
          else
          {
              ++mFrame;  // 프레임 증가

              if (mFrame == Anim->GetFrameCount())  // 마지막 프레임 초과
              {
                  if (mFinishNotify)
                      mFinishNotify();

                  if (mLoop)
                  {
                      mFrame = 0;  // 처음으로

                      // Notify 초기화
                      for (size_t i = 0; i < mNotifyArray.size(); ++i)
                          mNotifyArray[i].Call = false;
                  }
                  else
                      mFrame = Anim->GetFrameCount() - 1;  // 마지막 고정
              }
          }
      }

      // ★ Step 4: 프레임 이벤트 호출
      CallNotify();
  }

  8️⃣ 프레임별 이벤트 처리 (Animation2DSequence.cpp:131-146)

  void CAnimation2DSequence::CallNotify()
  {
      size_t Size = mNotifyArray.size();

      for (size_t i = 0; i < Size; ++i)
      {
          // 현재 프레임과 일치 && 아직 호출 안 됨
          if (mNotifyArray[i].Frame == mFrame && !mNotifyArray[i].Call)
          {
              mNotifyArray[i].Call = true;  // 중복 호출 방지

              if (mNotifyArray[i].Function)
                  mNotifyArray[i].Function();  // 콜백 실행
          }
      }
  }

  🔟 애니메이션 추가 (Animation2DComponent.cpp:57-99)

  void CAnimation2DComponent::AddAnimation(
      const std::weak_ptr<CAnimation2D>& Anim,
      float PlayTime, float PlayRate, bool Loop, bool Reverse)
  {
      auto _Anim = Anim.lock();
      if (!_Anim) return;

      // 이미 등록된 애니메이션인지 확인
      auto iter = mAnimationMap.find(_Anim->GetName());
      if (iter != mAnimationMap.end()) return;

      // ★ 새 Sequence 생성
      std::shared_ptr<CAnimation2DSequence> Sequence(new CAnimation2DSequence);

      Sequence->SetAnimation2D(Anim);
      Sequence->SetPlayTime(PlayTime);
      Sequence->SetPlayRate(PlayRate);
      Sequence->SetLoop(Loop);
      Sequence->SetReverse(Reverse);

      mAnimationMap.insert(std::make_pair(_Anim->GetName(), Sequence));

      // ★ 첫 애니메이션은 자동으로 Current로 설정
      if (!mCurrentAnimation)
      {
          mCurrentAnimation = Sequence;

          auto MeshComponent = mUpdateComponent.lock();
          if (MeshComponent)
          {
              auto Asset = mCurrentAnimation->GetAsset().lock();
              if (Asset)
              {
                  mAnimCBuffer->SetAnimation2DTextureType(Asset->GetAnimationTextureType());
                  MeshComponent->SetTexture(0, 0, Asset->GetTexture());
                  mUpdateEnable = true;
              }
          }
      }
  }

  1️⃣1️⃣ 애니메이션 전환 (Animation2DComponent.cpp:217-257)

  void CAnimation2DComponent::ChangeAnimation(const std::string& Name)
  {
      std::string Key = "Animation2D_" + Name;

      if (mUpdateComponent.expired()) return;
      if (!mCurrentAnimation) return;

      // 같은 애니메이션이면 무시
      if (mCurrentAnimation->GetName() == Key) return;

      auto iter = mAnimationMap.find(Key);
      if (iter == mAnimationMap.end()) return;

      // ★ 기존 애니메이션 초기화
      mCurrentAnimation->Clear();

      // ★ 새 애니메이션으로 전환
      mCurrentAnimation = iter->second;
      mCurrentAnimation->Clear();

      // ★ 텍스처 교체
      auto MeshComponent = mUpdateComponent.lock();
      if (MeshComponent)
      {
          auto Asset = mCurrentAnimation->GetAsset().lock();
          if (Asset)
          {
              mAnimCBuffer->SetAnimation2DTextureType(Asset->GetAnimationTextureType());
              MeshComponent->SetTexture(0, 0, Asset->GetTexture());
          }
      }
  }

  📡 PART 6: GPU로 UV 좌표 전송

  1️⃣2️⃣ SetShader 함수 (Animation2DComponent.cpp:259-293)

  void CAnimation2DComponent::SetShader()
  {
      int Frame = mCurrentAnimation->GetFrame();  // 현재 프레임 인덱스

      mAnimCBuffer->SetTextureSymmetry(mCurrentAnimation->GetSymmetry());

      auto Asset = mCurrentAnimation->GetAsset().lock();

      if (Asset)
      {
          if (Asset->GetAnimationTextureType() ==
              EAnimation2DTextureType::SpriteSheet)
          {
              // ★ 프레임 정보 가져오기
              const FTextureFrame& TexFrame = Asset->GetFrame(Frame);

              auto Texture = Asset->GetTexture().lock();

              if (Texture)
              {
                  const FTextureInfo* TexInfo = Texture->GetTexture();

                  // ★ 픽셀 좌표 → UV 좌표 변환
                  // UV 좌표는 0.0 ~ 1.0 범위
                  mAnimCBuffer->SetLTUV(
                      TexFrame.Start.x / TexInfo->Width,
                      TexFrame.Start.y / TexInfo->Height
                  );
                  mAnimCBuffer->SetRBUV(
                      (TexFrame.Start.x + TexFrame.Size.x) / TexInfo->Width,
                      (TexFrame.Start.y + TexFrame.Size.y) / TexInfo->Height
                  );
              }
          }
      }

      // ★ GPU로 전송
      mAnimCBuffer->UpdateBuffer();
  }

  UV 좌표 계산 예시:
  텍스처 크기: 1024x1024
  프레임 2: Start(256, 0), Size(128, 128)

  Left-Top UV:
    x = 256 / 1024 = 0.25
    y = 0 / 1024 = 0.0

  Right-Bottom UV:
    x = (256 + 128) / 1024 = 384 / 1024 = 0.375
    y = (0 + 128) / 1024 = 0.125

  → 셰이더는 UV(0.25, 0.0) ~ (0.375, 0.125) 영역만 샘플링

  🎯 핵심 알고리즘 정리

  프레임 전환 타이밍

  // 시간 기반 프레임 전환
  mTime += DeltaTime * mPlayRate;

  if (mTime >= mFrameTime)
  {
      mTime -= mFrameTime;  // ← 핵심! 남은 시간 보존
      ++mFrame;             // 정확한 타이밍 유지
  }

  왜 mTime = 0이 아니라 mTime -= mFrameTime?
  FrameTime = 0.125초
  DeltaTime = 0.020초 (50fps 일 때)

  Frame 0: mTime = 0.105
  다음: mTime = 0.105 + 0.020 = 0.125 (정확히 일치)
        → mTime = 0, Frame = 1

  Frame 1: mTime = 0.020
  다음: mTime = 0.020 + 0.020 = 0.040
  ...
  Frame 1: mTime = 0.120
  다음: mTime = 0.120 + 0.020 = 0.140 (초과!)
        → mTime = 0.140 - 0.125 = 0.015 (남은 시간 보존)
        → Frame = 2

  이렇게 하면 프레임 레이트와 무관하게 정확한 타이밍 유지!

  UV 좌표 변환

  픽셀 좌표 → UV 좌표 (0~1 정규화)

  LeftTop UV:
    u = FrameStartX / TextureWidth
    v = FrameStartY / TextureHeight

  RightBottom UV:
    u = (FrameStartX + FrameWidth) / TextureWidth
    v = (FrameStartY + FrameHeight) / TextureHeight

  auto anim = player->CreateComponent<CAnimation2DComponent>("Anim");

  anim->SetUpdateComponent(mesh);

  // 대기 애니메이션: 4프레임, 2초, 반복
  anim->AddAnimation("PlayerIdle", 2.0f, 1.0f, true, false);

  // 걷기 애니메이션: 8프레임, 0.8초, 반복
  anim->AddAnimation("PlayerWalk", 0.8f, 1.0f, true, false);

  // 공격 애니메이션: 6프레임, 0.6초, 반복 안 함
  anim->AddAnimation("PlayerAttack", 0.6f, 1.0f, false, false);

  // 이벤트 추가
  anim->AddNotify<CPlayer>("PlayerAttack", "DamageFrame", 3,
                           this, &CPlayer::DealDamage);

  // 게임 중 애니메이션 전환
  if (IsMoving())
      anim->ChangeAnimation("PlayerWalk");
  else
      anim->ChangeAnimation("PlayerIdle");

  if (IsAttacking())
      anim->ChangeAnimation("PlayerAttack");
