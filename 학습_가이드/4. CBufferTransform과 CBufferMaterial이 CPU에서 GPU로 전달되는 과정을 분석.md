  bool CConstantBuffer::Init(int Size, int Register, int ShaderBuffer)
  {
      mSize = Size;              // 버퍼 크기 (예: 324 bytes)
      mRegister = Register;      // 레지스터 번호 (0, 1, 2, ...)
      mShaderBuffer = ShaderBuffer;  // 어느 셰이더에서 사용? (VS, PS, ...)

      D3D11_BUFFER_DESC Desc = {};

      // ★ DYNAMIC 사용법 - 매 프레임 업데이트 가능
      Desc.Usage = D3D11_USAGE_DYNAMIC;       // CPU에서 자주 업데이트
      Desc.ByteWidth = mSize;                 // 버퍼 크기
      Desc.BindFlags = D3D11_BIND_CONSTANT_BUFFER;  // 상수 버퍼로 사용
      Desc.CPUAccessFlags = D3D11_CPU_ACCESS_WRITE; // CPU 쓰기 가능

      // GPU 메모리에 버퍼 생성
      CDevice::GetInst()->GetDevice()->CreateBuffer(&Desc, nullptr, &mBuffer);

      return true;
  }

  D3D11_USAGE_DYNAMIC의 장점:
  - CPU에서 매 프레임 업데이트 가능
  - Map/Unmap으로 효율적인 업데이트

  6️⃣ GPU로 데이터 전송 (CBufferTransform.cpp:31-48)

  void CCBufferTransform::UpdateBuffer()
  {
      // Step 1: 행렬 곱셈 계산
      mData.WV = mData.World * mData.View;
      mData.WVP = mData.WV * mData.Proj;

      // Step 2: 행렬 전치 (DirectX는 열 우선, HLSL은 행 우선)
      mData.World.Transpose();
      mData.View.Transpose();
      mData.Proj.Transpose();
      mData.WV.Transpose();
      mData.WVP.Transpose();

      // Step 3: 실제 GPU로 전송
      std::shared_ptr<CConstantBuffer> CBuffer = mBuffer.lock();
      if (CBuffer)
          CBuffer->Update(&mData);  // ← 여기서 전송!
  }

  7️⃣ 실제 GPU 전송 구현 (ConstantBuffer.cpp:36-90)

  void CConstantBuffer::Update(void* Data)
  {
      // ★★★ 핵심: Map/Unmap 패턴 ★★★

      // Step 1: GPU 버퍼를 CPU가 접근할 수 있게 잠금
      D3D11_MAPPED_SUBRESOURCE Map = {};
      CDevice::GetInst()->GetContext()->Map(
          mBuffer,                    // 대상 버퍼
          0,                          // 서브리소스 인덱스
          D3D11_MAP_WRITE_DISCARD,    // 기존 내용 버리고 새로 쓰기
          0,                          // 플래그
          &Map                        // 매핑된 주소 반환
      );

      // Step 2: CPU 메모리 → GPU 메모리 복사
      memcpy(Map.pData, Data, mSize);  // FCBufferTransformData 복사 (324 bytes)

      // Step 3: GPU 버퍼 잠금 해제
      CDevice::GetInst()->GetContext()->Unmap(mBuffer, 0);

      // Step 4: 셰이더 파이프라인에 상수 버퍼 바인딩

      // Vertex Shader에 바인딩
      if (mShaderBuffer & EShaderBufferType::Vertex)
      {
          CDevice::GetInst()->GetContext()->VSSetConstantBuffers(
              mRegister,   // 레지스터 번호 (0 = b0)
              1,           // 버퍼 개수
              &mBuffer     // 버퍼 포인터
          );
      }

      // Pixel Shader에 바인딩
      if (mShaderBuffer & EShaderBufferType::Pixel)
      {
          CDevice::GetInst()->GetContext()->PSSetConstantBuffers(
              mRegister,   // 레지스터 번호 (1 = b1)
              1,
              &mBuffer
          );
      }

      // Domain, Hull, Geometry, Compute Shader도 동일한 방식...
  }

  🔑 핵심 개념 정리

  Map/Unmap 패턴

  // 1. Map - GPU 메모리를 CPU가 접근 가능하게
  D3D11_MAPPED_SUBRESOURCE Map = {};
  Context->Map(buffer, 0, D3D11_MAP_WRITE_DISCARD, 0, &Map);

  // 2. 데이터 복사
  memcpy(Map.pData, &myData, size);

  // 3. Unmap - GPU가 다시 사용 가능하게
  Context->Unmap(buffer, 0);

  D3D11_MAP_WRITE_DISCARD:
  - 기존 데이터를 버리고 새로 씀
  - 가장 빠른 업데이트 방식
  - 매 프레임 전체를 갱신할 때 적합

  메모리 정렬 규칙

  DirectX 상수 버퍼는 16바이트 정렬 필수:

  // ❌ 잘못된 구조체
  struct Bad {
      float3 color;   // 12 bytes
      float opacity;  // 4 bytes
  };  // 총 16 bytes - OK처럼 보이지만...

  // ✅ 올바른 구조체
  struct Good {
      float4 color;   // 16 bytes (float3 + 1 padding)
      float opacity;  // 4 bytes
      float3 empty;   // 12 bytes (16바이트 경계 맞추기)
  };  // 총 32 bytes - 16의 배수!

  행렬 전치(Transpose)의 이유

  // DirectX (C++) - 행 우선(Row-major)
  FMatrix mat = {
      1, 2, 3, 4,
      5, 6, 7, 8,
      ...
  };

  // HLSL - 열 우선(Column-major)
  // 따라서 전송 전에 Transpose 필요!
  mat.Transpose();
