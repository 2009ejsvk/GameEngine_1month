  bool CGraphicShader::LoadVertexShader(const char* EntryName, ...)
  {
      // HLSL 파일을 vs_5_0 프로필로 컴파일
      D3DCompileFromFile(FullPath, nullptr,
          D3D_COMPILE_STANDARD_FILE_INCLUDE,
          EntryName,  // "Color2DVS"
          "vs_5_0",   // Shader Model 5.0
          Flag, 0, &mVSBlob, &ErrorBlob);

      // 컴파일된 바이트코드로 VertexShader 객체 생성
      CDevice::GetInst()->GetDevice()->CreateVertexShader(
          mVSBlob->GetBufferPointer(),
          mVSBlob->GetBufferSize(),
          nullptr, &mVS);
  }

  Vertex Shader 함수 (GlobalShader2D.hlsl:23-35)

  VS_OUTPUT_COLOR2D Color2DVS(VS_INPUT_COLOR2D input)
  {
      VS_OUTPUT_COLOR2D output = (VS_OUTPUT_COLOR2D)0;

      // 1. Pivot 보정 (중심점 조정)
      float3 Pos = input.Pos - cbPivotSize;

      // 2. 월드-뷰-투영 변환 (로컬 → 화면 좌표)
      output.Pos = mul(float4(Pos, 1.f), cbWVP);

      // 3. 색상 그대로 전달
      output.Color = input.Color;

      return output;
  }

  {
      PS_OUTPUT_COLOR output = (PS_OUTPUT_COLOR) 0;

      float4 TextureColor = tbBaseTexture.Sample(sbPoint, input.UV);

  🔄 전체 데이터 흐름 요약

  [CPU]
  FVertexColor (28 bytes)
  ├─ Pos: float3 (12 bytes)
  └─ Color: float4 (16 bytes)
      ↓ Vertex Buffer 생성

  [GPU Input Assembler]
  Input Layout 매핑
  ├─ POSITION0 ← Pos
  └─ COLOR0 ← Color
      ↓

  [Vertex Shader]
  VS_INPUT_COLOR2D
  ├─ Pos (float3)
  └─ Color (float4)
      ↓ cbWVP 행렬 변환
  VS_OUTPUT_COLOR2D
  ├─ Pos (SV_POSITION) → 화면 좌표
  └─ Color (COLOR) → 보간될 색상
      ↓ 래스터라이저 (삼각형 → 픽셀)

  [Pixel Shader]
  VS_OUTPUT_COLOR2D (보간된 값)
  └─ Color
      ↓
  PS_OUTPUT_COLOR
  └─ Color (SV_TARGET) → 백버퍼

  핵심 포인트

  1. 입력 레이아웃: AddInputDesc로 CPU 구조체와 HLSL 시맨틱을 연결
  2. VS 역할: 정점 변환 (로컬 → 월드 → 뷰 → 투영)
  3. PS 역할: 픽셀 색상 결정 (정점 색상 또는 텍스처 샘플링)
  4. 보간: VS 출력이 래스터라이저에서 자동으로 보간되어 PS 입력으로 전달
  5. 상수 버퍼: CPU에서 행렬/색상 등의 파라미터를 GPU로 전송
